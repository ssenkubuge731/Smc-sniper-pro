```cpp //+------------------------------------------------------------------+ //| SMC Sniper Pro APA - MT5 Expert Advisor | //| Advanced Price Action + Smart Money Concepts | //| Starter build - Demo testing only | //+------------------------------------------------------------------+ #property strict #property version "1.00" #include CTrade trade; enum TradingStyle { STYLE_SCALPING=0, STYLE_DAY=1, STYLE_SWING=2 }; input TradingStyle Mode = STYLE_DAY; input bool DemoOnly = true; input bool AllowTrading = false; input double FixedLots = 0.01; input int StopLossPoints = 300; input int TakeProfitPoints = 600; input int MaxOpenPositions = 1; input int MaxSpreadPoints = 50; input ulong MagicNumber = 7312026; input int SwingLookback = 20; input int FastEMA = 9; input int SlowEMA = 21; input int RSI_Period = 14; input double BuyRSIMin = 52.0; input double SellRSIMax = 48.0; input bool RequireLiquiditySweep = true; input bool RequireStructureBreak = true; input bool RequireOBorFVG = false; int hFast=INVALID_HANDLE; int hSlow=INVALID_HANDLE; int hRSI=INVALID_HANDLE; datetime lastBar=0; // Timeframes ENUM_TIMEFRAMES SignalTF() { if(Mode==STYLE_SCALPING) return PERIOD_M1; if(Mode==STYLE_DAY) return PERIOD_M15; return PERIOD_H1; } ENUM_TIMEFRAMES TrendTF() { if(Mode==STYLE_SCALPING) return PERIOD_M15; if(Mode==STYLE_DAY) return PERIOD_H1; return PERIOD_D1; } // New candle detection bool NewBar() { datetime t=iTime(_Symbol,SignalTF(),0); if(t!=lastBar && t>0) { lastBar=t; return true; } return false; } // Indicator data double Buf(int handle,int shift) { double a[]; ArraySetAsSeries(a,true); if(handle==INVALID_HANDLE) return EMPTY_VALUE; if(CopyBuffer(handle,0,shift,1,a)!=1) return EMPTY_VALUE; return a[0]; } // Spread filter bool SpreadOK() { long sp=SymbolInfoInteger(_Symbol,SYMBOL_SPREAD); return sp>0 && sp<=MaxSpreadPoints; } // Count EA positions int CountPositions() { int n=0; for(int i=PositionsTotal()-1;i>=0;i--) { ulong ticket=PositionGetTicket(i); if(ticket>0 && PositionGetString(POSITION_SYMBOL)==_Symbol && (ulong)PositionGetInteger(POSITION_MAGIC)==MagicNumber) n++; } return n; } // Demo protection bool IsDemoAccount() { ENUM_ACCOUNT_TRADE_MODE m= (ENUM_ACCOUNT_TRADE_MODE)AccountInfoInteger(ACCOUNT_TRADE_MODE); return m==ACCOUNT_TRADE_MODE_DEMO; } // Bullish liquidity sweep bool BullSweep() { double low1=iLow(_Symbol,SignalTF(),1); double close1=iClose(_Symbol,SignalTF(),1); double ref=DBL_MAX; for(int i=2;i<=SwingLookback+1;i++) ref=MathMin(ref,iLow(_Symbol,SignalTF(),i)); return low1ref; } // Bearish liquidity sweep bool BearSweep() { double high1=iHigh(_Symbol,SignalTF(),1); double close1=iClose(_Symbol,SignalTF(),1); double ref=-DBL_MAX; for(int i=2;i<=SwingLookback+1;i++) ref=MathMax(ref,iHigh(_Symbol,SignalTF(),i)); return high1>ref && close1ref; } // Bearish structure break bool BearBreak() { double c=iClose(_Symbol,SignalTF(),1); double ref=DBL_MAX; for(int i=2;i<=SwingLookback+1;i++) ref=MathMin(ref,iLow(_Symbol,SignalTF(),i)); return cBuf(hSlow,1); } bool TrendBear() { return Buf(hFast,1) iOpen(_Symbol,SignalTF(),1); } bool BearCandle() { return iClose(_Symbol,SignalTF(),1)< iOpen(_Symbol,SignalTF(),1); } // Approximate OB/FVG filters bool BullZoneApprox() { return iClose(_Symbol,SignalTF(),1)> iHigh(_Symbol,SignalTF(),2); } bool BearZoneApprox() { return iClose(_Symbol,SignalTF(),1)< iLow(_Symbol,SignalTF(),2); } // BUY conditions bool BuySignal() { bool sweep=!RequireLiquiditySweep || BullSweep(); bool brk=!RequireStructureBreak || BullBreak(); bool zone=!RequireOBorFVG || BullZoneApprox(); return TrendBull() && sweep && brk && zone && BullCandle() && Buf(hRSI,1)>=BuyRSIMin; } // SELL conditions bool SellSignal() { bool sweep=!RequireLiquiditySweep || BearSweep(); bool brk=!RequireStructureBreak || BearBreak(); bool zone=!RequireOBorFVG || BearZoneApprox(); return TrendBear() && sweep && brk && zone && BearCandle() && Buf(hRSI,1)<=SellRSIMax; } // Normalize lot size double NormalizeLots(double lots) { double minv=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MIN); double maxv=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MAX); double step=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_STEP); lots=MathMax(minv,MathMin(maxv,lots)); lots=MathFloor(lots/step)*step; return NormalizeDouble(lots,2); } // BUY order void OpenBuy() { double ask=SymbolInfoDouble(_Symbol,SYMBOL_ASK); double sl=ask-StopLossPoints*_Point; double tp=ask+TakeProfitPoints*_Point; trade.SetExpertMagicNumber(MagicNumber); trade.Buy( NormalizeLots(FixedLots), _Symbol, ask, sl, tp, "SMC APA BUY" ); } // SELL order void OpenSell() { double bid=SymbolInfoDouble(_Symbol,SYMBOL_BID); double sl=bid+StopLossPoints*_Point; double tp=bid-TakeProfitPoints*_Point; trade.SetExpertMagicNumber(MagicNumber); trade.Sell( NormalizeLots(FixedLots), _Symbol, bid, sl, tp, "SMC APA SELL" ); } // Initialize EA int OnInit() { hFast=iMA( _Symbol,TrendTF(),FastEMA,0,MODE_EMA,PRICE_CLOSE ); hSlow=iMA( _Symbol,TrendTF(),SlowEMA,0,MODE_EMA,PRICE_CLOSE ); hRSI=iRSI( _Symbol,SignalTF(),RSI_Period,PRICE_CLOSE ); if(hFast==INVALID_HANDLE || hSlow==INVALID_HANDLE || hRSI==INVALID_HANDLE) return INIT_FAILED; trade.SetExpertMagicNumber(MagicNumber); return INIT_SUCCEEDED; } // Release indicators void OnDeinit(const int reason) { if(hFast!=INVALID_HANDLE) IndicatorRelease(hFast); if(hSlow!=INVALID_HANDLE) IndicatorRelease(hSlow); if(hRSI!=INVALID_HANDLE) IndicatorRelease(hRSI); } // Main trading engine void OnTick() { if(!NewBar()) return; if(DemoOnly && !IsDemoAccount()) return; if(!AllowTrading) return; if(!SpreadOK()) return; if(CountPositions()>=MaxOpenPositions) return; if(BuySignal()) OpenBuy(); else if(SellSignal()) OpenSell(); } ```
